            function removeTabStops(elements, originalTabindexAttribute) {
                Array.prototype.forEach.call(elements, function (element) {
                    if (
                        originalTabindexAttribute &&
                        !element.hasAttribute(originalTabindexAttribute)
                    ) {
                        var originalTabindex = element.getAttribute("tabindex");
                        element.setAttribute(originalTabindexAttribute, originalTabindex === null ? "" : originalTabindex);
                    }

                    if (element.getAttribute("tabindex") !== "-1") {
                        element.setAttribute("tabindex", "-1");
                    }
                });
            }

            function restoreTabStops(elements, originalTabindexAttribute) {
                Array.prototype.forEach.call(elements, function (element) {
                    if (!element.hasAttribute(originalTabindexAttribute)) {
                        return;
                    }

                    var originalTabindex = element.getAttribute(originalTabindexAttribute);

                    if (originalTabindex === "") {
                        element.removeAttribute("tabindex");
                    } else {
                        element.setAttribute("tabindex", originalTabindex);
                    }

                    element.removeAttribute(originalTabindexAttribute);
                });
            }

            function removeCookieScriptTabStops() {
                var cookieDialog = document.getElementById("cookiescript_injected");

                if (!cookieDialog) {
                    return;
                }

                var tabStops = cookieDialog.querySelectorAll("[tabindex]");

                if (cookieDialog.hasAttribute("tabindex")) {
                    tabStops = Array.prototype.slice.call(tabStops);
                    tabStops.unshift(cookieDialog);

                removeTabStops(tabStops);

                    basmPanel.removeAttribute("aria-hidden");
                    basmPanel.removeAttribute("inert");
                    restoreTabStops(
                        basmPanel.querySelectorAll("[" + originalTabindexAttribute + "]"),
                        originalTabindexAttribute
                    );
                    return;
                }

                basmPanel.setAttribute("aria-hidden", "true");
                basmPanel.setAttribute("inert", "");

                var tabStops = basmPanel.querySelectorAll("a, button, input, select, textarea, iframe, [tabindex], [role='button']");

                if (basmPanel.hasAttribute("tabindex")) {
                    tabStops = Array.prototype.slice.call(tabStops);
                    tabStops.unshift(basmPanel);
