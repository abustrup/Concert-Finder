    <script type="text/javascript">
  // Deletes legacy consent/analytics cookies left behind by the removed
  // OneTrust + Adobe Launch setup. Runs at most once per browser: once the
  // GUARD cookie has been set the block exits immediately, so cookies set
  // by GA/Freestar/etc. after this script runs are never touched.
  //
  // To force everyone through the purge again (e.g. after another cleanup),
  // bump the GUARD name to `sk_cookie_purge_v2`, `_v3`, etc.
  (function () {
    var GUARD = 'sk_cookie_purge_v1';
    // Guard clause: if this browser has already been purged for this version,
    // do nothing. Bumping GUARD is what re-triggers the purge for everyone.
    if (document.cookie.indexOf(GUARD + '=') !== -1) return;

    // Allowlist of first-party cookies to KEEP. Anything not listed here (or
    // matched by KEEP_PREFIX) gets expired. HttpOnly cookies (_skweb_session,
    // auth_http_s) are already invisible to document.cookie and cannot be
    // touched by JS - they're listed defensively.
    var KEEP = {
      '_skweb_session': 1,
      'auth_http': 1,
      'auth_http_s': 1,
      'session_uuid': 1,
      'variant_cohort': 1,
      'override_split_test': 1,
      'filter_metro_area': 1,
      'dismissed_discovery_tooltip': 1,
      'view_desktop_app_cta': 1,
      'view_attendance_popup': 1,
      'num_calendar_events_post_signup': 1,
      'post_authentication': 1,
      'postsignup_imported_artist_count': 1,
      'postsignup_taste_import_source': 1,
      'ticket_vendor_link': 1,
      'show_ticket_link_popup': 1,
      'analytics_trigger': 1,
      'sk_disable_admin': 1
    };
    var KEEP_PREFIX = ['taste_imports_success_'];

    // document.cookie doesn't expose a cookie's original domain/path, so to
    // delete reliably we re-issue the expiry against every plausible
    // domain/path combo. Includes the bare host, dotted parent domains, and
    // the shared .songkick.com / .songkick.net roots.
    var host = window.location.hostname;
    var domains = ['', host, '.' + host, '.songkick.com', '.songkick.net'];
    var parts = host.split('.');
    for (var i = 0; i < parts.length - 1; i++) {
      domains.push('.' + parts.slice(i).join('.'));
    }
    var paths = ['/', window.location.pathname];

    function keep(name) {
      if (name === GUARD || KEEP[name]) return true;
      for (var p = 0; p < KEEP_PREFIX.length; p++) {
        if (name.indexOf(KEEP_PREFIX[p]) === 0) return true;
      }
      return false;
    }

    var rawCookies = document.cookie ? document.cookie.split(';') : [];
    for (var c = 0; c < rawCookies.length; c++) {
      var raw = rawCookies[c];
      var eq = raw.indexOf('=');
      var name = (eq > -1 ? raw.slice(0, eq) : raw).replace(/^\s+/, '');
      if (!name || keep(name)) continue;
      var base = name + '=; expires=Thu, 01 Jan 1970 00:00:00 GMT';
      for (var d = 0; d < domains.length; d++) {
        for (var pth = 0; pth < paths.length; pth++) {
          document.cookie = base + '; path=' + paths[pth] + (domains[d] ? '; domain=' + domains[d] : '');
        }
      }
    }

    // Set the guard last so a mid-run failure doesn't leave a browser marked
    // as purged. 5-year max-age matches the practical retention we want.
    document.cookie = GUARD + '=1; path=/; max-age=' + (60 * 60 * 24 * 365 * 5);
  })();
</script>
